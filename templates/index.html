async function createMusic() { let userPrompt = document.getElementById('promptInput').value; // جو پرامپٹ صارف لکھے let response = await fetch('/generate-music', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ prompt: userPrompt }) }); let data = await response.json(); if(data.success) { console.audio("گانا تیار ہے:", data.audio_url); // یہاں آڈیو پلے کرنے کا کوڈ لگا دیں } else { alert("کوئی خرابی ہو گئی: " + data.error); } }
