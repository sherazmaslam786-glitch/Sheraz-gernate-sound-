// یہ فنکشن کسی بھی ٹیکسٹ کو لے کر اسے فورا آواز میں بولے گا (Text-to-Speech Engine) function speakText(textToSpeak) { if ('speechSynthesis' in window) { // پرانی کوئی آواز اگر چل رہی ہو تو اسے روک دیں window.speechSynthesis.cancel(); let utterance = new SpeechSynthesisUtterance(textToSpeak); utterance.lang = 'ur-PK'; // اردو زبان کے لیے (اگر انگلینڈ یا دوسری ہو تو en-US بھی کر سکتے ہیں) utterance.rate = 1.0; // بولنے کی رفتار utterance.pitch = 1.0; وائس پچ window.speechSynthesis.speak(utterance); } else { alert("معذرت، آپ کا براؤزر اسپیچ (Speech) کو سپورٹ نہیں کرتا۔"); } } // مثال کے طور پر جب کوئی یوزر گانا یا ٹیکسٹ پلے کرے تو یہ فنکشن کال کریں: function playSongOrText() { let userText = "جنید سراج اور جنید ساؤنڈ اسٹوڈیو میں خوش آمدید۔ آپ کا گانا تیار ہے۔"; speakText(userText); }
